Configure Secure Supabase RLS

Run this SQL once in Supabase SQL Editor so customers can browse products & submit orders while only logged-in Admin (authenticated) can edit parts or read customer records: 

CREATE UNIQUE INDEX IF NOT EXISTS customers_mobile_number_key ON public.customers (mobile_number);

-- 1. PRODUCT: Public read, Admin (authenticated) write/delete only
ALTER TABLE public.product ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Allow all on product" ON public.product;
DROP POLICY IF EXISTS "Public read products" ON public.product;
DROP POLICY IF EXISTS "Admin manage products" ON public.product;
CREATE POLICY "Public read products" ON public.product FOR SELECT TO anon, authenticated USING (true);
CREATE POLICY "Admin manage products" ON public.product FOR ALL TO authenticated USING (true) WITH CHECK (true);
GRANT SELECT ON TABLE public.product TO anon;
GRANT ALL ON TABLE public.product TO authenticated;

-- 2. CUSTOMERS: Checkout upsert allowed, Admin-only read/delete (protects PII)
ALTER TABLE public.customers ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Allow all on customers" ON public.customers;
DROP POLICY IF EXISTS "Checkout insert customers" ON public.customers;
DROP POLICY IF EXISTS "Checkout update customers" ON public.customers;
DROP POLICY IF EXISTS "Admin read customers" ON public.customers;
DROP POLICY IF EXISTS "Admin delete customers" ON public.customers;
CREATE POLICY "Checkout insert customers" ON public.customers FOR INSERT TO anon, authenticated WITH CHECK (true);
CREATE POLICY "Checkout update customers" ON public.customers FOR UPDATE TO anon, authenticated USING (true) WITH CHECK (true);
CREATE POLICY "Admin read customers" ON public.customers FOR SELECT TO authenticated USING (true);
CREATE POLICY "Admin delete customers" ON public.customers FOR DELETE TO authenticated USING (true);
GRANT INSERT, UPDATE ON TABLE public.customers TO anon;
GRANT ALL ON TABLE public.customers TO authenticated;

-- 3. CUSTOMER_ORDERS: Checkout insert allowed, Admin-only read/manage
ALTER TABLE public.customer_orders ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Allow all on customer_orders" ON public.customer_orders;
DROP POLICY IF EXISTS "Checkout insert orders" ON public.customer_orders;
DROP POLICY IF EXISTS "Admin manage orders" ON public.customer_orders;
CREATE POLICY "Checkout insert orders" ON public.customer_orders FOR INSERT TO anon, authenticated WITH CHECK (true);
CREATE POLICY "Admin manage orders" ON public.customer_orders FOR SELECT TO authenticated USING (true);
CREATE POLICY "Admin update orders" ON public.customer_orders FOR UPDATE TO authenticated USING (true) WITH CHECK (true);
CREATE POLICY "Admin delete orders" ON public.customer_orders FOR DELETE TO authenticated USING (true);
GRANT INSERT ON TABLE public.customer_orders TO anon;
GRANT ALL ON TABLE public.customer_orders TO authenticated;

GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO anon, authenticated;
